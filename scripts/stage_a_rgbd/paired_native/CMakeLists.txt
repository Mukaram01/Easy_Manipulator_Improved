cmake_minimum_required(VERSION 3.16)
project(workcell_paired_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(PkgConfig REQUIRED)
find_package(ignition-gazebo6 6.18.0 EXACT REQUIRED)
find_package(ignition-gazebo6-rendering REQUIRED)
pkg_check_modules(RENDER REQUIRED IMPORTED_TARGET ignition-rendering6 ignition-utils1)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET jsoncpp)
pkg_check_modules(OGRE2 REQUIRED IMPORTED_TARGET ignition-rendering6-ogre2 OGRE-Next)
pkg_check_modules(GL REQUIRED IMPORTED_TARGET gl)
add_executable(stage_a_paired_native capture.cpp)
target_link_libraries(stage_a_paired_native ignition-gazebo6::ignition-gazebo6 ignition-gazebo6::ignition-gazebo6-rendering PkgConfig::RENDER PkgConfig::JSON PkgConfig::OGRE2 PkgConfig::GL)
add_executable(stage_a_paired_frozen_test test_frozen.cpp)
target_link_libraries(stage_a_paired_frozen_test ignition-gazebo6::ignition-gazebo6 ignition-gazebo6::ignition-gazebo6-rendering PkgConfig::RENDER PkgConfig::JSON PkgConfig::OGRE2 PkgConfig::GL)
target_compile_options(stage_a_paired_frozen_test PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME paired_frozen_synthetic COMMAND stage_a_paired_frozen_test)
# Offline arithmetic probe; never constructs a renderer or simulator.
add_executable(stage_a_paired_ray_replay ray_replay.cpp)
target_link_libraries(stage_a_paired_ray_replay PkgConfig::OGRE2 PkgConfig::JSON)
target_compile_options(stage_a_paired_ray_replay PRIVATE -O2)
