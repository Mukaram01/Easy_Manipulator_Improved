cmake_minimum_required(VERSION 3.16)
project(workcell_stage_a_rgbd LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Standalone prerequisite probe: does not link EPD or launch a simulator.
option(STAGE_A_RENDER_PROBE_ONLY "Build only the simulation identity capability probe" OFF)
if(STAGE_A_RENDER_PROBE_ONLY)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(RENDER REQUIRED IMPORTED_TARGET ignition-rendering6 ignition-utils1)
  pkg_check_modules(JSON REQUIRED IMPORTED_TARGET jsoncpp)
  add_executable(stage_a_render_identity_probe render_identity_probe.cpp)
  target_link_libraries(stage_a_render_identity_probe PkgConfig::RENDER PkgConfig::JSON)
  pkg_check_modules(OGRE2 REQUIRED IMPORTED_TARGET ignition-rendering6-ogre2 OGRE-Next)
  pkg_check_modules(GL REQUIRED IMPORTED_TARGET gl)
  add_executable(stage_a_ogre2_image_probe render_identity_probe.cpp)
  target_compile_definitions(stage_a_ogre2_image_probe PRIVATE STAGE_A_OGRE2_IMAGES=1)
  target_link_libraries(stage_a_ogre2_image_probe PkgConfig::RENDER PkgConfig::JSON PkgConfig::OGRE2 PkgConfig::GL)
  add_executable(stage_a_fragment_mrt fragment_mrt.cpp)
  target_link_libraries(stage_a_fragment_mrt PkgConfig::RENDER PkgConfig::JSON PkgConfig::OGRE2 PkgConfig::GL)
  option(STAGE_A_GAZEBO_FRAGMENT_CAPTURE "Build disposable live Gazebo MRT capture owner" OFF)
  if(STAGE_A_GAZEBO_FRAGMENT_CAPTURE)
    find_package(ignition-gazebo6 6.18.0 EXACT REQUIRED)
    find_package(ignition-gazebo6-rendering REQUIRED)
    add_executable(stage_a_gazebo_fragment_capture gazebo_fragment_capture.cpp)
    target_link_libraries(stage_a_gazebo_fragment_capture ignition-gazebo6::ignition-gazebo6
      ignition-gazebo6::ignition-gazebo6-rendering PkgConfig::RENDER PkgConfig::JSON PkgConfig::OGRE2 PkgConfig::GL)
  endif()
  add_executable(stage_a_gl_context_test gl_context_test.cpp)
  target_link_libraries(stage_a_gl_context_test PkgConfig::JSON)
  target_compile_options(stage_a_gl_context_test PRIVATE -UNDEBUG)
  add_test(NAME gl_context_classification COMMAND stage_a_gl_context_test)
  add_executable(stage_a_renderer_identity_test renderer_identity_test.cpp)
  target_link_libraries(stage_a_renderer_identity_test PkgConfig::JSON)
  add_test(NAME renderer_identity_diagnostics COMMAND stage_a_renderer_identity_test)
  add_executable(stage_a_inventory_compare_test inventory_compare_test.cpp)
  target_link_libraries(stage_a_inventory_compare_test PkgConfig::JSON)
  add_test(NAME inventory_lossless_comparison COMMAND stage_a_inventory_compare_test)
  add_executable(stage_a_compositor_uniform_test test_compositor_uniform.cpp)
  target_link_libraries(stage_a_compositor_uniform_test PkgConfig::OGRE2)
  target_compile_options(stage_a_compositor_uniform_test PRIVATE -UNDEBUG)
  enable_testing()
  add_test(NAME compositor_uniform_storage COMMAND stage_a_compositor_uniform_test)
  return()
endif()
set(EPD_SOURCE_DIR "" CACHE PATH "External easy_perception_deployment package source (read only)")
if(NOT EXISTS "${EPD_SOURCE_DIR}/src/p3_ort_base.cpp")
  message(FATAL_ERROR "Set EPD_SOURCE_DIR to the external EPD package source")
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(epd_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(tf2 REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(epd_onnxruntime_vendor REQUIRED)
find_package(OpenCV REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
find_package(PkgConfig REQUIRED)
pkg_check_modules(IGN REQUIRED IMPORTED_TARGET ignition-transport11 ignition-msgs8)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET jsoncpp)
pkg_check_modules(XML REQUIRED IMPORTED_TARGET tinyxml2)
add_executable(stage_a_rgbd_capture capture.cpp
  "${EPD_SOURCE_DIR}/src/ort_base.cpp" "${EPD_SOURCE_DIR}/src/p3_ort_base.cpp")
add_executable(stage_a_rgbd_geometry_test test_geometry.cpp)
foreach(target stage_a_rgbd_capture stage_a_rgbd_geometry_test)
  target_include_directories(${target} PRIVATE "${EPD_SOURCE_DIR}/include" ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
  ament_target_dependencies(${target} rclcpp sensor_msgs epd_msgs cv_bridge tf2 pcl_conversions)
  target_link_libraries(${target} ${OpenCV_LIBS} ${PCL_LIBRARIES})
  target_compile_definitions(${target} PRIVATE USE_GPU=false PRINT_MODEL_INFO=false EPD_ENABLE_TENSORRT=0)
endforeach()
target_link_libraries(stage_a_rgbd_capture PkgConfig::XML PkgConfig::IGN PkgConfig::JSON epd_onnxruntime_vendor::onnxruntime)
target_link_libraries(stage_a_rgbd_geometry_test PkgConfig::XML PkgConfig::JSON)
target_compile_options(stage_a_rgbd_geometry_test PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME masked_depth_geometry COMMAND stage_a_rgbd_geometry_test)
